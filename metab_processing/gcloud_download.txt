gcloud config set storage/check_hashes never

DEST=/global/scratch/users/fosterangus/MetabTravLR/Data/MERSCOPE/vz-ffpe-showcase
for d in $(gcloud storage ls gs://vz-ffpe-showcase/); do
  name=$(basename "$d")
  mkdir -p "$DEST/$name"
  echo ">>> $name"
  gcloud storage cp -n "${d}cell_by_gene.csv" "${d}cell_metadata.csv" "$DEST/$name/"
done

Then verify you got all 18 of each:
ls "$DEST"/*/cell_by_gene.csv  | wc -l     # expect 18
ls "$DEST"/*/cell_metadata.csv | wc -l     # expect 18
du -sh "$DEST"                             # expect ~30 GB

If any folder is missing a file, the wc -l will be < 18 â re-running the same block (it's idempotent with -n) will grab whatever's missing.
